cmake_minimum_required(VERSION 3.16)
project(dynarmic_native_bench LANGUAGES C CXX)

# This is deliberately a separate project. The repository root builds the Wasm
# backend even with a native compiler (for emitter tests), not native Dynarmic.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
get_filename_component(DYNARMIC_JS_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(DYNARMIC_JS_SOURCE_ROOT "${DYNARMIC_JS_SOURCE_ROOT}" CACHE INTERNAL "Benchmark source checkout")
if (EMSCRIPTEN OR CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native comparison currently supports native Linux x86_64 and arm64 hosts")
endif()
if (DEFINED ARCHITECTURE)
    message(FATAL_ERROR "Do not override ARCHITECTURE; the native comparator detects its host backend")
endif()
include("${DYNARMIC_JS_SOURCE_ROOT}/externals/dynarmic/CMakeModules/DetectArchitecture.cmake")
if (NOT ARCHITECTURE STREQUAL "x86_64" AND NOT ARCHITECTURE STREQUAL "arm64")
    message(FATAL_ERROR "The native comparison supports only x86_64 and arm64 backends")
endif()
set(DYNARMIC_NATIVE_BACKEND "${ARCHITECTURE}" CACHE INTERNAL "Detected native backend")
add_subdirectory("${DYNARMIC_JS_SOURCE_ROOT}/externals/dynarmic" upstream)

enable_testing()
foreach(host jit_bench jit_thumb_bench jit_a64_bench jit_a64_simd_bench jit_callbench)
    add_executable(${host} "${DYNARMIC_JS_SOURCE_ROOT}/tests/hosts/${host}.cpp")
    target_link_libraries(${host} PRIVATE dynarmic merry::mcl)
endforeach()
foreach(host jit_bench jit_thumb_bench jit_a64_bench)
    foreach(mode 0 1 2)
        add_test(NAME ${host}_${mode} COMMAND ${host} ${mode})
    endforeach()
endforeach()
add_test(NAME jit_callbench COMMAND jit_callbench)
add_test(NAME jit_a64_neon COMMAND jit_a64_simd_bench neon)
add_test(NAME jit_a64_scalar COMMAND jit_a64_simd_bench scalar)
get_property(benchmark_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${benchmark_tests} PROPERTIES LABELS "benchmark;native" TIMEOUT 60)
